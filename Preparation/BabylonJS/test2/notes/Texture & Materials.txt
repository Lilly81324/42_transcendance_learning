Using Materials
1) Make a Material
	const material_name = new BABYLON.StandardMaterial("name", scene);
2) Texture the Material:
	- material_name.diffuseTexture = new BABYLON.Texture(...)
2.5) Color the Material:
	- Unicolor:
		material_name.diffuseColor = new BABYLON.Color3(0, 1, 0);
		(Keep in mind that you have to subtract color values.
		1, 1, 1 is default lighting, and if you want it half green,
		you will have to do 0.5, 1, 1)
3) Assign Material to Object
	ground.material = material_name;

Textures:
	new BABYLON.Texture("<file url>", scene);

UV Wrapping Textures:
Using 1 Texture for an image is stupid and restrictive.
Unlesss you specify, which part of a texture should be displayed.
If you have an Image, which contains multiple sections,
you can assign each section to a face, by specifying a vector for the area.

1) Create array of Vectors
	The Vectors range from bottom left of the picture (0, 0) to top right (1, 1).
	The first two vectors are the bottom x and y of the pic.
	So the Vector4(0, 0, 0.5, 1) goes from bottom left,
	to the middle width, and all the way to the top of the image.
	const faceUv = [];
	faceUv[0] = new BABYLON.Vector4(0.5, 0.0, 0.75, 1.0); //rear face
	faceUv[1] = new BABYLON.Vector4(0.0, 0.0, 0.25, 1.0); //front face
	faceUv[2] = new BABYLON.Vector4(0.25, 0, 0.5, 1.0); //right side
	faceUv[3] = new BABYLON.Vector4(0.75, 0, 1.0, 1.0); //left side
	(I have no idea how the order is decided, or how more complex shapes handle this)
2) Create Object, set faceUV to our faceUv vector, and enable Wrapping
	const box = BABYLON.MeshBuilder.CreateBox("box", {faceUV: fUV, wrap: true});

function buildRoof(position = new BABYLON.Vector3.Zero(), scene)
{
	// Create Material for Roof
	const material = new BABYLON.StandardMaterial("Roof Material", scene);

	// Create Texture for Base
	material.diffuseTexture = new BABYLON.Texture("https://assets.babylonjs.com/textures/mercator.jpg", scene);

	// Map Texture to Roof
	const uvMap = [];
	uvMap[0] = new BABYLON.Vector4(0, 0, 0.5, 0.5);
	uvMap[1] = new BABYLON.Vector4(0, 0.5, 0.5, 1);
	uvMap[2] = new BABYLON.Vector4(0.5, 0, 1, 0.5);
	uvMap[3] = new BABYLON.Vector4(0.5, 0.5, 1, 1);

	// Create Roof
	const roof = new BABYLON.MeshBuilder.CreateCylinder("roof", {diameter: 1.5, height: 1.0, tessellation: 3, faceUV: uvMap, wrap: true});
	roof.scaling.x = 0.6;
	roof.rotation.z = Math.PI / 2;
	roof.position = position;
	roof.position.y = position.y + 1.22;
}

 * 
 * @param position BABYLON.Vector3 type that points to current position
 */
export function buildHouse(position = new BABYLON.Vector3.Zero(), scene)
{
	// Create Material for Base
	const material = new BABYLON.StandardMaterial("House Material", scene);
	
	// Create Texture for Base
	material.diffuseTexture = new BABYLON.Texture("https://doc.babylonjs.com/img/resources/textures_thumbs/skybox3.jpg", scene);

	// Map Texture for Base
	let uvMap = [];
	uvMap[0] = new BABYLON.Vector4(0, (1/3), 0.25, (2/3));
	uvMap[1] = new BABYLON.Vector4(0.25, (1/3), 0.5, (2/3));
	uvMap[2] = new BABYLON.Vector4(0.5, (1/3), 0.75, (2/3));
	uvMap[3] = new BABYLON.Vector4(0.75, (1/3), 1, (2/3));
	uvMap[4] = new BABYLON.Vector4(0.25, (2/3), 0.5, 1);
	uvMap[5] = new BABYLON.Vector4(0.25, 0, 0.5, (1/3));

	// Create Box with given Mapping and Material
	const box = new BABYLON.MeshBuilder.CreateBox("box", {faceUV: uvMap, wrap: true}, scene)
    box.position = position;
	box.material = material;

	buildRoof(position, scene);
}
