function everyFrame(scene, projectiles, inputs) {
    const magnitude = 0.1;
    const map_boundary = 5;
    const speed = 0.05;
    const rot_speed = 0.025;
    const player = scene.getMeshByName("player");
    scene.actionManager = new BABYLON.ActionManager();
    scene.actionManager.registerAction(new BABYLON.ExecuteCodeAction(
        BABYLON.ActionManager.OnEveryFrameTrigger,
        () => {
            if (inputs.left)
                player.rotation.z = player.rotation.z + rot_speed;
            if (inputs.right)
                player.rotation.z = player.rotation.z - rot_speed;
            if (inputs.up)
                player.position.x = player.position.x + speed;
            if (inputs.down)
                player.position.x = player.position.x - speed;
            for (var i = 0; i < projectiles.length; i++) {
                var proj = projectiles[i]
                if (proj.position.x > map_boundary || proj.position.x < -map_boundary || proj.position.y > map_boundary || proj.position.y < -map_boundary)
                {
                    proj.dispose();
                    projectiles.splice(i, 1);
                    i--;
                    continue ;
                }
                var direction = angleToVector(proj.rotation.z);
                proj.position.x += direction[0] * magnitude;
                proj.position.y += direction[1] * magnitude;
            }
        }
    ));
}

function createPlayer(scene, ground, inputs) {
    // Create Mesh
    var box1 = BABYLON.MeshBuilder.CreateBox("player", {size: 0.5, height: 0.7,}, scene);
    var sphere = BABYLON.MeshBuilder.CreateSphere("sphere", {diameter: 0.5, segments: 10}, scene);
    sphere.position.y = 0.35;
    var player = BABYLON.Mesh.MergeMeshes([box1, sphere], true, false, null, false, false);
    player.position.y = 2;
    player.name = "player";

    scene.onKeyboardObservable.add((kbInfo) => {
        // Is Button down or up?
        let setBool = kbInfo.type == BABYLON.KeyboardEventTypes.KEYDOWN ? true : false;

        if (kbInfo.type == BABYLON.KeyboardEventTypes.KEYDOWN || 
            kbInfo.type == BABYLON.KeyboardEventTypes.KEYUP) {
            if (kbInfo.event.key == "a")
                inputs.left = setBool;
            if (kbInfo.event.key == "d")
                inputs.right = setBool;
            if (kbInfo.event.key == "w")
                inputs.up = setBool;
            if (kbInfo.event.key == "s")
                inputs.down = setBool;
        }
    });
    player.actionManager = new BABYLON.ActionManager();
    player.actionManager.registerAction(new BABYLON.ExecuteCodeAction(
        {
            trigger: BABYLON.ActionManager.OnIntersectionEnterTrigger,
            parameter: {
                mesh: ground,
                usePreciseIntersection: true,
            },
        },
        () => { console.log("Hello World");}
    ));
    return (player);
}

// In radians
function angleToVector(angle) {
    var result = [-Math.sin(angle), Math.cos(angle)];
    return (result);
}

function registerProjectiles(scene, player) {
    var projectiles = [];
    const magnitude = 0.4;
    var proj_counter = 0;
    scene.onKeyboardObservable.add((kbInfo) => {
    if (kbInfo.type == BABYLON.KeyboardEventTypes.KEYDOWN &&
        kbInfo.event.key == " ")
        {
            const new_proj = BABYLON.MeshBuilder.CreateSphere("projectile_" + proj_counter, {segments: 10, diameter: 0.3}, scene);
            const directionVector = angleToVector(player.rotation.z);
            new_proj.rotation.z = player.rotation.z;
            directionVector[0] *= magnitude;
            directionVector[1] *= magnitude;
            new_proj.position = new BABYLON.Vector3(
                player.position.x + directionVector[0],
                player.position.y + directionVector[1],
                0
            )
            projectiles.push(new_proj);
            proj_counter++;
        }
    });
    return (projectiles);
}

function createGround(scene, player)
{
    const bottom = -2;
    const depth = 1;
    // Move counter clockwise
    const array = [
        new BABYLON.Vector3(5, 0, bottom),
        new BABYLON.Vector3(5, 0, 0),
        new BABYLON.Vector3(4, 0, 1),
        new BABYLON.Vector3(-4, 0, 1),
        new BABYLON.Vector3(-5, 0, 0),
        new BABYLON.Vector3(-5, 0, bottom)
    ];
    const ground = BABYLON.MeshBuilder.ExtrudePolygon("Ground_Box", {shape: array, depth: depth}, scene);
	ground.rotation.x = -Math.PI / 2;
	ground.position = new BABYLON.Vector3(0, 0, -depth / 2);

	//const floorAggregate = new BABYLON.PhysicsAggregate(ground, BABYLON.PhysicsShapeType.MESH, { mass: 0 }, scene);

    return (ground);
}

export const createScene = function () {
    // This creates a basic Babylon Scene object (non-mesh)
    var scene = new BABYLON.Scene(engine);
    scene.actionManager = new BABYLON.ActionManager(scene);

    // This creates and positions a free camera (non-mesh)
    var camera = new BABYLON.FreeCamera("camera1", new BABYLON.Vector3(0, 5, -10), scene);

    // This targets the camera to scene origin
    camera.setTarget(BABYLON.Vector3.Zero());

    // This attaches the camera to the canvas
    camera.attachControl(canvas, true);

    // This creates a light, aiming 0,1,0 - to the sky (non-mesh)
    var light = new BABYLON.HemisphericLight("light", new BABYLON.Vector3(0, 1, 0), scene);

    // Default intensity is 1. Let's dim the light a small amount
    light.intensity = 0.7;

    // Our built-in 'ground' shape.
    var ground = createGround(scene);

    var inputs = {left: false, right: false, up: false, down: false};
    var player = createPlayer(scene, ground, inputs);
    var projectiles = registerProjectiles(scene, player);
    everyFrame(scene, projectiles, inputs);

    return scene;
};
